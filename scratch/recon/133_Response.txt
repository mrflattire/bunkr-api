<script>
async function getSignedUrl(rawUrl) {
    const orig = new URL(rawUrl);
    
    
    const path = decodeURIComponent(orig.pathname);
    const r = await fetch(
        signUrl + '?path=' + encodeURIComponent(path),
        { credentials: 'omit' }
    );
    if (!r.ok) throw new Error('sign ' + r.status);
    const { token, ex } = await r.json();
    orig.searchParams.set('token', token);
    orig.searchParams.set('ex', ex);
    return orig.toString();
}

function buildPlayer(srcUrl, type, poster) {
    const wrapper = document.getElementById('video-wrapper');
    const video   = document.createElement('video');
    video.id      = 'player';
    video.controls = true;
    video.preload  = 'none';
    video.setAttribute('class', '--plyr-color-main: #a78bfa;');
    if (poster) video.setAttribute('data-poster', poster);

    const source = document.createElement('source');
    source.src  = srcUrl;
    source.type = type || 'video/mp4';
    video.appendChild(source);
    wrapper.appendChild(video);

    document.getElementById('loading-background').style.display = 'none';
    wrapper.style.display = 'block';

    return new Plyr(video, {
        debug: false,
        tooltips: { controls: true },
        controls: ['play-large', 'play', 'progress', 'current-time', 'mute', 'volume', 'settings', 'pip', 'airplay', 'fullscreen'],
        autoplay: false,
        ratio: '16:9'
    });
}

            console.error('Error loading more files:', error);
            button.disabled = false;
            document.getElementById('loading').classList.add('hidden');
        });
});
